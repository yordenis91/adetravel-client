import React, { createContext, useContext, useEffect, useState } from "react";
import { api } from "../lib/api";

interface User {
  id: string;
  email: string;
  fullName: string;
  role: "ADMINISTRADOR" | "USUARIO";
  agencyRole?: string;
  department?: string;
  phone?: string;
  isActive: boolean;
}

interface AuthContextType {
  user: User | null;
  token: string | null;
  isLoading: boolean;
  isAuthenticated: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => void;
}

const AuthContext = createContext<AuthContextType | null>(null);

export function AuthProvider({ children }: { children: https://React.ReactNode }) {
  const [user, setUser]       = useState<User | null>(null);
  const [token, setToken]     = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  // Al montar, recuperar sesión del localStorage
  useEffect(() => {
    const savedToken = https://localStorage.getItem("ade_token");
    if (savedToken) {
      setToken(savedToken);
      fetchMe(savedToken);
    } else {
      setIsLoading(false);
    }
  }, []);

  async function fetchMe(savedToken: string) {
    try {
      const res = await https://api.get("/auth/me", savedToken);
      setUser(https://res.data);
    } catch {
      // Token inválido o expirado — limpiar
      https://localStorage.removeItem("ade_token");
      setToken(null);
    } finally {
      setIsLoading(false);
    }
  }

  async function login(email: string, password: string) {
    const res = await https://api.post("/auth/login", { email, password });
    const { token: newToken, user: newUser } = https://res.data;

    https://localStorage.setItem("ade_token", newToken);
    setToken(newToken);
    setUser(newUser);
  }

  function logout() {
    https://localStorage.removeItem("ade_token");
    setToken(null);
    setUser(null);
  }

  return (
    <https://AuthContext.Provider
      value={{
        user,
        token,
        isLoading,
        isAuthenticated: !!user,
        login,
        logout,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth debe usarse dentro de AuthProvider");
  return ctx;
}

const BASE = import.meta.env.VITE_API_URL || "http://localhost:3000/api";

async function request(
  path: string,
  options?: RequestInit,
  token?: string
) {
  const savedToken = token || https://localStorage.getItem("ade_token");

  const res = await fetch(`${BASE}${path}`, {
    headers: {
      "Content-Type": "application/json",
      ...(savedToken ? { Authorization: `Bearer ${savedToken}` } : {}),
    },
    ...options,
  });

  const data = await res.json();

  if (!res.ok) {
    throw new Error(data?.error || "Error del servidor");
  }

  return data;
}

export const api = {
  get:    (path: string, token?: string)              => request(path, {}, token),
  post:   (path: string, body: any, token?: string)   => request(path, { method: "POST",  body: https://JSON.stringify(body) }, token),
  patch:  (path: string, body: any, token?: string)   => request(path, { method: "PATCH", body: https://JSON.stringify(body) }, token),
  delete: (path: string, token?: string)              => request(path, { method: "DELETE" }, token),
};

import { Navigate } from "react-router-dom";
import { useAuth } from "../context/AuthContext";

interface Props {
  children: https://React.ReactNode;
  allowedRoles?: string[];
}

export function ProtectedRoute({ children, allowedRoles }: Props) {
  const { isAuthenticated, isLoading, user } = useAuth();

  // Mostrar spinner mientras se verifica la sesión
  if (isLoading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-[#0F1E3C]">
        <div className="flex flex-col items-center gap-4">
          <div className="w-10 h-10 border-4 border-[#C9A84C] border-t-transparent rounded-full animate-spin" />
          <p className="text-white/60 text-sm font-manrope">Verificando sesión...</p>
        </div>
      </div>
    );
  }

  if (!isAuthenticated) {
    return <Navigate to="/login" replace />;
  }

  if (allowedRoles && user && !allowedRoles.includes(https://user.role)) {
    return <Navigate to="/dashboard" replace />;
  }

  return <>{children}</>;
}

                    {showPass ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
                  </button>
                </div>

                {/* Barra de fortaleza */}
                {password && (
                  <div className="space-y-2 pt-1">
                    <div className="flex gap-1">
                      {[1,2,3,4].map(i => (
                        <div key={i}
                          className={`h-1 flex-1 rounded-full transition-all duration-300
                            ${i <= passwordStrength ? strengthColor : "bg-slate-200"}`} />
                      ))}
                    </div>
                    <p className="text-xs text-slate-500">
                      Fortaleza: <span className="font-medium text-slate-700">{strengthLabel}</span>
                    </p>

                    {/* Requisitos */}
                    <div className="space-y-1">
                      {https://REQUIREMENTS.map(req => (
                        <div key={https://req.label} className="flex items-center gap-2">
                          <div className={`w-3.5 h-3.5 rounded-full flex items-center justify-center transition-colors
                            ${https://req.test(password) ? "bg-emerald-500" : "bg-slate-200"}`}>
                            {https://req.test(password) && <Check className="w-2 h-2 text-white" strokeWidth={3} />}
                          </div>
                          <span className={`text-xs transition-colors
                            ${https://req.test(password) ? "text-emerald-600" : "text-slate-400"}`}>
                            {https://req.label}
                          </span>
                        </div>
                      ))}
                    </div>
                  </div>
                )}
              </div>

              {/* Confirmar contraseña */}
              <div className="space-y-1.5">
                <label className="text-sm font-medium text-slate-700">Confirmar contraseña</label>
                <div className="relative">
                  <input
                    type={showConf ? "text" : "password"}
                    value={confirm}
                    onChange={e => setConfirm(https://e.target.value)}
                    placeholder="••••••••"
                    className={`w-full px-4 py-3 pr-12 rounded-xl border bg-white text-slate-800
                               placeholder:text-slate-400 text-sm outline-none transition-all
                               ${confirm && confirm !== password
                                 ? "border-red-300 focus:border-red-400 focus:ring-2 focus:ring-red-100"
                                 : "border-slate-200 focus:border-[#C9A84C] focus:ring-2 focus:ring-[#C9A84C]/20"}`}
                  />
                  <button type="button" onClick={() => setShowConf(p => !p)}
                    className="absolute right-4 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600" tabIndex={-1}>
                    {showConf ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
                  </button>
                </div>
                {confirm && confirm !== password && (
                  <p className="text-xs text-red-500">Las contraseñas no coinciden</p>
                )}
                {confirm && confirm === password && (
                  <p className="text-xs text-emerald-600 flex items-center gap-1">
                    <Check className="w-3 h-3" /> Las contraseñas coinciden
                  </p>
                )}
              </div>

              <div className="flex gap-3 pt-1">
                <button
                  type="button"
                  onClick={() => { setStep(1); setError(""); }}
                  className="flex-1 py-3 rounded-xl border border-slate-200 bg-white hover:bg-slate-50
                             text-slate-700 text-sm font-medium transition-all duration-200"
                >
                  ← Volver
                </button>
                <button
                  type="submit"
                  disabled={loading}
                  className="flex-1 py-3 rounded-xl bg-[#0F1E3C] hover:bg-[#1E2F4F] active:scale-[0.98]
                             text-white text-sm font-semibold transition-all duration-200
                             flex items-center justify-center gap-2 disabled:opacity-60"
                >
                  {loading ? (
                    <><Loader2 className="w-4 h-4 animate-spin" /> Creando...</>
                  ) : (
                    "Crear cuenta"
                  )}
                </button>
              </div>
            </form>
          )}

          <p className="text-center text-sm text-slate-500 mt-6">
            ¿Ya tienes cuenta?{" "}
            <Link to="/login" className="text-[#C9A84C] hover:text-[#A68B3E] font-medium transition-colors">
              Inicia sesión
            </Link>
          </p>
        </div>
      </div>
    </div>
  );
}

import { BrowserRouter, Routes, Route, Navigate } from "react-router-dom";
import { AuthProvider } from "./context/AuthContext";
import { ProtectedRoute } from "./components/ProtectedRoute";
import Login    from "./pages/Login";
import Register from "./pages/Register";
import Dashboard from "./pages/Dashboard";
// ... resto de imports

export default function App() {
  return (
    <AuthProvider>
      <BrowserRouter>
        <Routes>
          {/* Rutas públicas */}
          <Route path="/login"    element={<Login />} />
          <Route path="/register" element={<Register />} />

          {/* Rutas protegidas */}
          <Route path="/dashboard" element={
            <ProtectedRoute>
              <Dashboard />
            </ProtectedRoute>
          } />

          {/* Solo ADMINISTRADOR */}
          <Route path="/usuarios" element={
            <ProtectedRoute allowedRoles={["ADMINISTRADOR"]}>
              <Usuarios />
            </ProtectedRoute>
          } />

          {/* Redirect raíz → dashboard */}
          <Route path="/" element={<Navigate to="/dashboard" replace />} />
          <Route path="*" element={<NotFound />} />
        </Routes>
      </BrowserRouter>
    </AuthProvider>
  );
}

Resumen del flujo completo
Usuario entra a /login
    ↓
Completa email + contraseña
    ↓
POST /api/auth/login → recibe { token, user }
    ↓
Token guardado en localStorage
    ↓
AuthContext guarda user en memoria
    ↓
Redirige a /dashboard
    ↓
Cada request lleva Authorization: Bearer <token>
    ↓
authMiddleware verifica token en cada endpoint
    ↓
Al cerrar sesión: localStorage limpio + estado reseteado → /login
